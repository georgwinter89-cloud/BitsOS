# BitsOS V2 CONFIG.TXT defaults written by `xtask image` (format: name=value, last wins, '#' starts a comment).
# Extra lines from `--config-extra "k=v;k2=v2"` and from ktest (`ktest=<filter>`) are appended below these.

# Kernel log verbosity: error, warn, info, debug, trace
loglevel=info
# What the screen console shows (IN-quiet): console_log=off|error|warn|info|debug|all (default warn = Warn+ plus
# boot, link, DHCP, MCP, watchdog, init, reload, power, input lines). loglevel= decides the ring and the mirror.
# Desktop (DS-4): the desktop starts after boot with the boot screen once the GPU is up (QEMU has none, so QEMU keeps
# the text console; the Gaming-PC keeps it too until its GPU driver is done).
# The desktop is \BITSOS2\bin\bitscomp. Any key during the boot screen shows the text console. Test only:
# ktest_desktop_program=<path> [args] starts a stand-in instead (host/tests/in/50-boot-screen.sh).
# What a kernel panic does after it printed: halt (stay for the serial log) or reboot
panic=halt
# Console mirror (RB-4): mirror=a.b.c.d[:port] sends every log line as UDP (port 6666 if left out). Test only:
# rb_mirror_panic=on forces a panic ktest (QEMU).
# Display (DP-1): display_refresh=<Hz>, e.g. 59.95, replaces the 60 Hz vblank estimate until EDID exists.
# Data volume (FS-8c): data_img_mb=<MiB> size of a new \BITSOS2\data.img (default 2048, 32..4095; 0 = no /data).
# An existing image is never changed; a new one is made only when 512 MiB of the stick stay free.
# The desktop (\BITSOS2\bin\bitscomp, linked against the Mesa tree \BITSOS2\mesa\) draws only on the GPU, with GLES,
# all effects and the shader cache; it has no key here. Without a GPU render node it ends with one line and the text
# console stays; a lost GPU ends it the same way.
# GPU at boot (GX-8c): the Radeon 780M comes up by itself at every boot (video included), within 30 s; the desktop
# waits for it. A failure, or a machine without the chip: no desktop, the text console says so in one line.
# Test only: drm=fake gives QEMU a fake GPU, drm_fake_hang=lose makes a hang on it end the device as an
# unrecoverable one does on the chip, gpu_boot_fake=up|fail|hang plays the bring-up under a hypervisor.
