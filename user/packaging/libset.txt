# UL-5: the glibc library set for dynamic programs, copied from CachyOS (the build host) into every image.
# <soname> <when>: always = in every image (programs copied later, over MCP or onto the stick, find them);
# needed = only when a program of the image needs it (ldd). A library ldd names for a program and not listed
# here joins as "needed" too. The files land in /usr/lib as on CachyOS, the loader too (/lib64 and /lib are links
# to usr/lib, /usr/lib64 to lib); glibc's only system directory there; V2 has no /etc/ld.so.cache.
# Versions, sizes, SHA-256 and licences of the set: libset.lock (cargo xtask initrd --lock rewrites it).
ld-linux-x86-64.so.2 always
libc.so.6            always
libm.so.6            always
# glibc >= 2.34 has libpthread, libdl and librt inside libc; these are its empty stubs for older programs.
libpthread.so.0      always
libdl.so.2           always
librt.so.1           always
libgcc_s.so.1        always
# Always (UL-5 fixer): C++ programs put on the stick or written over MCP later (Node, Brave, PK-4) need it.
libstdc++.so.6       always
# INT-W27b (SX-3c CR1): the rest of CachyOS's glibc package (1.5 MB), so a runtime that brings its own libraries on
# top of the host's glibc finds every one (pressure-vessel's libgio links libresolv) and name services work as there.
libanl.so.1          always
libBrokenLocale.so.1 always
libc_malloc_debug.so.0 always
libmvec.so.1         always
libnsl.so.1          always
libnss_compat.so.2   always
libnss_db.so.2       always
libnss_dns.so.2      always
